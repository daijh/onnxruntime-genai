# Dependencies for run_z_image_turbo.py / run_z_image_turbo_legacy.py (the inference drivers)
# ONLY, with the WebGPU execution provider. Separate from requirements.txt (the
# export_models.py / build_*.py side) since the run scripts never import onnxruntime_genai,
# onnx, torch's export machinery, or diffusers -- just plain onnxruntime plus a small
# numpy/PIL/transformers inference pipeline.
#
# The public PyPI `onnxruntime` wheel has no WebGPU execution provider built in (e.g.
# `onnxruntime==1.29.0` only exposes CPUExecutionProvider/AzureExecutionProvider).
# `onnxruntime-webgpu` is a separate, self-contained PyPI distribution that ships the same
# `onnxruntime` top-level package with WebGPU baked directly in -- `get_available_providers()`
# reports WebGpuExecutionProvider immediately after install, no plug-in registration call or
# extra package index needed. It's a *replacement* for `onnxruntime`, not an addition: don't
# install both `onnxruntime` and `onnxruntime-webgpu` in the same environment (or venv) as
# they both own the `onnxruntime` import path and will clobber each other.
onnxruntime-webgpu==1.27.0

numpy==2.5.3

# CPU-only build: the default PyPI `torch` wheel bundles/depends on nvidia-*-cu12 CUDA
# packages even though these run scripts drive inference through onnxruntime-webgpu, not
# torch. `--extra-index-url` pulls the `+cpu` variant from PyTorch's own index instead.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.14.0+cpu
psutil==7.2.2
pillow==12.3.0
transformers==5.16.1
